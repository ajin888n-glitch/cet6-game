    function initAudio() {
      if (!audioCtx) {
        const AudioContextClass = window.AudioContext || window.webkitAudioContext;
        if (AudioContextClass) audioCtx = new AudioContextClass();
      }
      if (audioCtx && audioCtx.state === 'suspended') {
        audioCtx.resume();
      }
    }

    function playTone(freq, type, duration, delay = 0) {
      if (!userStats.soundEnabled) return;
      try {
        initAudio();
        if (!audioCtx) return;
        setTimeout(() => {
          const osc = audioCtx.createOscillator();
          const gain = audioCtx.createGain();
          osc.type = type;
          osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
          gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
          gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
          osc.connect(gain);
          gain.connect(audioCtx.destination);
          osc.start();
          osc.stop(audioCtx.currentTime + duration);
        }, delay * 1000);
      } catch (e) {}
    }

    function soundSuccess() {
      playTone(523.25, 'triangle', 0.1, 0);
      playTone(659.25, 'triangle', 0.1, 0.08);
      playTone(783.99, 'triangle', 0.18, 0.15);
    }

    function soundError() {
      playTone(280, 'sawtooth', 0.15, 0);
      playTone(210, 'sawtooth', 0.2, 0.1);
    }

    function soundPop() {
      playTone(480, 'sine', 0.05, 0);
    }

    function soundCelebration() {
      [523.25, 659.25, 783.99, 1046.50].forEach((f, i) => {
        playTone(f, 'sine', 0.2, i * 0.08);
      });
    }

    function speakWord(text) {
      if (!('speechSynthesis' in window)) return;
      window.speechSynthesis.cancel();
      const utter = new SpeechSynthesisUtterance(text);
      utter.lang = 'en-US';
      utter.rate = 0.88;
      window.speechSynthesis.speak(utter);
    }

      } else if (currentGameMode === 'spelling' && spellingTargetWord) {
        speakWord(spellingTargetWord.word);
      }
    }

    function startRushGame() {
      currentGameMode = 'rush';
      const pool = getFilteredWords();
      gameWordQueue = shuffleArray(pool).slice(0, 10);
      currentWordIndex = 0;
      currentScore = 0;

    function loadRushQuestion() {
      clearInterval(rushTimer);
      if (currentWordIndex >= gameWordQueue.length) {
        finishGame();
        return;
      }

      const item = gameWordQueue[currentWordIndex];
      document.getElementById('rushProgress').innerText = `${currentWordIndex + 1}/${gameWordQueue.length}`;
      document.getElementById('rushWord').innerText = item.word;
      document.getElementById('rushPos').innerText = item.pos;
      document.getElementById('rushPhonetic').innerText = item.phonetic;

      speakWord(item.word);

      const optionsContainer = document.getElementById('rushOptionsGrid');
      optionsContainer.innerHTML = '';

      const distractorPool = CET6_DATABASE.filter(w => w.id !== item.id);
      const wrongItems = shuffleArray(distractorPool).slice(0, 3);
      const options = shuffleArray([item, ...wrongItems]);

      options.forEach(opt => {
        const btn = document.createElement('button');
        btn.className = 'w-full text-left px-3.5 py-3 rounded-xl bg-slate-900 hover:bg-slate-800 text-slate-100 border border-slate-800 font-semibold text-xs transition-all flex items-center justify-between touch-btn';
        btn.innerHTML = `
          <span class="truncate pr-2">${opt.meaning}</span>
          <i class="fa-regular fa-circle text-slate-500 text-[10px] flex-shrink-0"></i>
        `;
        btn.onclick = () => handleRushAnswer(opt.id === item.id, btn, item);
        optionsContainer.appendChild(btn);
      });

      startRushTimer();
    }

    function startRushTimer() {
      rushTimeLeft = 100;
      const bar = document.getElementById('rushTimeBar');
      bar.style.width = '100%';
      bar.className = 'h-full bg-amber-400 w-full transition-all duration-100 ease-linear';

      const intervalMs = 100;
      const totalDuration = 8000;
      const decrement = (intervalMs / totalDuration) * 100;

      rushTimer = setInterval(() => {
        rushTimeLeft -= decrement;
        if (rushTimeLeft <= 0) {
          clearInterval(rushTimer);
          bar.style.width = '0%';
          handleRushAnswer(false, null, gameWordQueue[currentWordIndex]);
        } else {
          bar.style.width = `${rushTimeLeft}%`;
          if (rushTimeLeft < 30) {
            bar.className = 'h-full bg-rose-500 w-full transition-all duration-100 ease-linear';
          }
        }
      }, intervalMs);
    }

    function handleRushAnswer(isCorrect, btnElem, currentWord) {
      clearInterval(rushTimer);
      const options = document.querySelectorAll('#rushOptionsGrid button');
      options.forEach(b => b.disabled = true);

      if (isCorrect) {
        soundSuccess();
        currentCombo++;
        if (currentCombo > maxCombo) maxCombo = currentCombo;
        const speedBonus = Math.floor(rushTimeLeft / 12);
        currentScore += (20 + (currentCombo - 1) * 5 + speedBonus);

        document.getElementById('rushScore').innerText = currentScore;
        document.getElementById('rushCombo').innerText = currentCombo;

        if (btnElem) {
          btnElem.classList.remove('bg-slate-900', 'border-slate-800');
          btnElem.classList.add('bg-emerald-950', 'border-emerald-500', 'text-emerald-200');
          btnElem.querySelector('i').className = 'fa-solid fa-check text-emerald-400 text-xs';
        }
      } else {
        soundError();
        currentCombo = 0;
        document.getElementById('rushCombo').innerText = '0';
        addToNotebook(currentWord);

        if (btnElem) {
          btnElem.classList.remove('bg-slate-900', 'border-slate-800');
          btnElem.classList.add('bg-rose-950', 'border-rose-500', 'text-rose-200', 'animate-shake');
          btnElem.querySelector('i').className = 'fa-solid fa-xmark text-rose-400 text-xs';
        }

        options.forEach(b => {
          if (b.innerText.includes(currentWord.meaning)) {
            b.classList.add('border-emerald-500', 'bg-emerald-950/70', 'text-emerald-300');
          }
        });
      }

      setTimeout(() => {
        currentWordIndex++;
        loadRushQuestion();
      }, 900);
    }

    function startSpellingGame() {
      currentGameMode = 'spelling';
      const pool = getFilteredWords();
      // 筛选适合手机屏宽展示的词长 (<= 10个字母)
      const validPool = pool.filter(w => w.word.length <= 10);
      gameWordQueue = shuffleArray(validPool).slice(0, 8);
      currentWordIndex = 0;
      currentScore = 0;

      document.getElementById('homeView').classList.add('hidden');
      document.getElementById('spellingView').classList.remove('hidden');
      document.getElementById('spellScore').innerText = '0';

      loadSpellingRound();
    }

    function loadSpellingRound() {
      if (currentWordIndex >= gameWordQueue.length) {
        finishGame();
        return;
      }

      spellingTargetWord = gameWordQueue[currentWordIndex];
      userSpelledLetters = [];

      document.getElementById('spellProgress').innerText = `${currentWordIndex + 1}/${gameWordQueue.length}`;
      document.getElementById('spellMeaning').innerText = spellingTargetWord.meaning;
      document.getElementById('spellExample').innerText = `"${spellingTargetWord.example}"`;

      speakWord(spellingTargetWord.word);

      const letters = spellingTargetWord.word.toUpperCase().split('');
      availableBankLetters = shuffleArray(letters.map((char, idx) => ({ id: idx, char: char, used: false })));

      renderSpellingUI();
    }

    function renderSpellingUI() {
      const targetLen = spellingTargetWord.word.length;
      const slotsContainer = document.getElementById('spellSlots');
      slotsContainer.innerHTML = '';

      // 动态适配字符大小与槽位宽
      const slotSizeClass = targetLen >= 9 ? 'w-8 h-10 text-base' : 'w-9 h-11 text-lg';

      for (let i = 0; i < targetLen; i++) {
        const slot = document.createElement('div');
        const filled = userSpelledLetters[i];
        slot.className = `${slotSizeClass} rounded-xl flex items-center justify-center font-game font-black border-2 transition-all ${
          filled ? 'bg-cyan-950 border-cyan-400 text-cyan-300 shadow scale-105' : 'bg-slate-900 border-dashed border-slate-700 text-slate-600'
        }`;
        slot.innerText = filled ? filled.char : '_';
        slotsContainer.appendChild(slot);
      }

      const bankContainer = document.getElementById('spellBank');
      bankContainer.innerHTML = '';

      const bankSizeClass = targetLen >= 9 ? 'w-8 h-10 text-base' : 'w-9 h-11 text-lg';

      availableBankLetters.forEach(item => {
        const bubble = document.createElement('button');
        bubble.className = `${bankSizeClass} rounded-xl font-game font-black flex items-center justify-center border transition touch-btn ${
          item.used
            ? 'opacity-20 cursor-not-allowed bg-slate-900 border-slate-800 text-slate-600'
            : 'bg-slate-800 hover:bg-cyan-700 text-white border-slate-700 active:border-cyan-400'
        }`;
        bubble.innerText = item.char;
        bubble.disabled = item.used;
        bubble.onclick = () => pickLetter(item);
        bankContainer.appendChild(bubble);
      });
    }

    function pickLetter(item) {
      if (item.used) return;
      soundPop();
      item.used = true;
      userSpelledLetters.push(item);
      renderSpellingUI();

      if (userSpelledLetters.length === spellingTargetWord.word.length) {
        checkSpellingResult();
      }
    }

    function undoLastLetter() {
      if (userSpelledLetters.length === 0) return;
      soundPop();
      const removed = userSpelledLetters.pop();
      const bankItem = availableBankLetters.find(b => b.id === removed.id);
      if (bankItem) bankItem.used = false;
      renderSpellingUI();
    }

    function resetSpellingWord() {
      soundPop();
      userSpelledLetters = [];
      availableBankLetters.forEach(b => b.used = false);
      renderSpellingUI();
    }

    function giveSpellingHint() {
      const cleanWord = spellingTargetWord.word.toUpperCase();
      const nextIndex = userSpelledLetters.length;
      if (nextIndex >= cleanWord.length) return;

      const neededChar = cleanWord[nextIndex];
      const matchInBank = availableBankLetters.find(b => !b.used && b.char === neededChar);

      if (matchInBank) {
        currentScore = Math.max(0, currentScore - 10);
        document.getElementById('spellScore').innerText = currentScore;
        pickLetter(matchInBank);
      }
    }

    function checkSpellingResult() {
      const target = spellingTargetWord.word.toUpperCase();
      const input = userSpelledLetters.map(l => l.char).join('');

      if (input === target) {
        soundSuccess();
        currentScore += 30;
        document.getElementById('spellScore').innerText = currentScore;

        const slots = document.querySelectorAll('#spellSlots div');
        slots.forEach(s => {
          s.classList.remove('border-cyan-400');
          s.classList.add('border-emerald-400', 'bg-emerald-950', 'text-emerald-300');
        });

        setTimeout(() => {
          currentWordIndex++;
          loadSpellingRound();
        }, 900);
      } else {
        soundError();
        addToNotebook(spellingTargetWord);

        const slots = document.querySelectorAll('#spellSlots div');
        slots.forEach(s => s.classList.add('animate-shake', 'border-rose-400', 'bg-rose-950'));

        setTimeout(() => {
          resetSpellingWord();
        }, 800);
      }
    }

    // 模式3: 消消配对
    function startMatchGame() {
      currentGameMode = 'match';
      const pool = getFilteredWords();
      const selectedWords = shuffleArray(pool).slice(0, 6);

      currentScore = 0;
      matchedPairsCount = 0;
      matchMovesCount = 0;
      firstSelectedCard = null;
      secondSelectedCard = null;
      matchLocked = false;

      document.getElementById('homeView').classList.add('hidden');
      document.getElementById('matchView').classList.remove('hidden');
      document.getElementById('matchPairsCount').innerText = `0/6`;
      document.getElementById('matchMoves').innerText = '0';

      matchCards = [];
      selectedWords.forEach((wordObj, i) => {
        matchCards.push({
          uid: `en_${i}`,
          wordId: wordObj.id,
          text: wordObj.word,
          type: 'en',
          rawObj: wordObj,
          matched: false
        });
        matchCards.push({
          uid: `cn_${i}`,
          wordId: wordObj.id,
          text: wordObj.meaning,
          type: 'cn',
          rawObj: wordObj,
          matched: false
        });
      });

      matchCards = shuffleArray(matchCards);
      renderMatchGrid();
    }

    function renderMatchGrid() {
      const grid = document.getElementById('matchGrid');
      grid.innerHTML = '';

      matchCards.forEach(card => {
        const cardElem = document.createElement('div');
        cardElem.id = `card_${card.uid}`;
        cardElem.className = `h-20 rounded-xl flex flex-col items-center justify-center text-center p-1.5 cursor-pointer border transition-all duration-200 select-none touch-btn ${
          card.matched
            ? 'opacity-0 pointer-events-none scale-75'
            : 'bg-slate-900 border-slate-800 hover:border-amber-400/80 shadow'
        }`;

        if (card.type === 'en') {
          cardElem.innerHTML = `
            <span class="text-[9px] text-amber-400 font-mono mb-0.5">EN</span>
            <span class="text-xs font-black text-white font-game break-all line-clamp-1">${card.text}</span>
          `;
        } else {
          cardElem.innerHTML = `
            <span class="text-[9px] text-emerald-400 font-mono mb-0.5">释义</span>
            <span class="text-[11px] font-semibold text-slate-200 line-clamp-2 leading-tight">${card.text}</span>
          `;
        }

        cardElem.onclick = () => handleCardClick(card, cardElem);
        grid.appendChild(cardElem);
      });
    }

    function handleCardClick(card, cardElem) {
      if (matchLocked || card.matched) return;
      if (firstSelectedCard && firstSelectedCard.card.uid === card.uid) return;

      soundPop();
      if (card.type === 'en') speakWord(card.rawObj.word);

      cardElem.classList.remove('border-slate-800');
      cardElem.classList.add('border-amber-400', 'bg-amber-400/10');

      if (!firstSelectedCard) {
        firstSelectedCard = { card, elem: cardElem };
      } else {
        secondSelectedCard = { card, elem: cardElem };
        matchMovesCount++;
        document.getElementById('matchMoves').innerText = matchMovesCount;
        evaluatePair();
      }
    }

    function evaluatePair() {
      matchLocked = true;
      const c1 = firstSelectedCard.card;
      const c2 = secondSelectedCard.card;

      const isMatch = c1.wordId === c2.wordId && c1.type !== c2.type;

      if (isMatch) {
        soundSuccess();
        matchedPairsCount++;
        document.getElementById('matchPairsCount').innerText = `${matchedPairsCount}/6`;

        firstSelectedCard.elem.classList.add('border-emerald-400', 'bg-emerald-950');
        secondSelectedCard.elem.classList.add('border-emerald-400', 'bg-emerald-950');

        setTimeout(() => {
          c1.matched = true;
          c2.matched = true;
          firstSelectedCard.elem.classList.add('opacity-0', 'scale-75');
          secondSelectedCard.elem.classList.add('opacity-0', 'scale-75');
          resetCardSelection();

          if (matchedPairsCount >= 6) {
            currentScore = Math.max(100, 240 - (matchMovesCount - 6) * 12);
            setTimeout(finishGame, 500);
          }
        }, 450);
      } else {
        soundError();
        addToNotebook(c1.rawObj);
        addToNotebook(c2.rawObj);

        firstSelectedCard.elem.classList.add('animate-shake', 'border-rose-400');
        secondSelectedCard.elem.classList.add('animate-shake', 'border-rose-400');

        setTimeout(() => {
          firstSelectedCard.elem.classList.remove('border-amber-400', 'bg-amber-400/10', 'animate-shake', 'border-rose-400');
          firstSelectedCard.elem.classList.add('border-slate-800');
          secondSelectedCard.elem.classList.remove('border-amber-400', 'bg-amber-400/10', 'animate-shake', 'border-rose-400');
          secondSelectedCard.elem.classList.add('border-slate-800');
          resetCardSelection();
        }, 600);
      }
    }

    function resetCardSelection() {
      firstSelectedCard = null;
      secondSelectedCard = null;
      matchLocked = false;
    }

    function restartMatchGame() {
      soundPop();
      startMatchGame();
    }

    function finishGame() {
      clearInterval(rushTimer);
      soundCelebration();
      triggerConfetti();

      const earnedXp = Math.floor(currentScore / 4) + 15;
      gainXP(earnedXp);

      document.getElementById('finalScore').innerText = currentScore;
      document.getElementById('earnedXp').innerText = `+${earnedXp} XP`;
      document.getElementById('finalMaxCombo').innerText = maxCombo > 0 ? maxCombo : '-';

      document.getElementById('resultModal').classList.remove('hidden');
    }

    function replayCurrentGame() {
      document.getElementById('resultModal').classList.add('hidden');
      if (currentGameMode === 'rush') startRushGame();
      else if (currentGameMode === 'spelling') startSpellingGame();
      else if (currentGameMode === 'match') startMatchGame();
    }

    function gainXP(amount) {
      userStats.xp += amount;
      const neededXP = userStats.level * 80;
      if (userStats.xp >= neededXP) {
        userStats.xp -= neededXP;
        userStats.level++;
        soundCelebration();
      }
      updateStatsUI();
    }

    function updateStatsUI() {
      document.getElementById('userLevel').innerText = userStats.level;
      const neededXP = userStats.level * 80;
      const percent = Math.min(100, Math.floor((userStats.xp / neededXP) * 100));
      document.getElementById('xpProgressBar').style.width = `${percent}%`;
      document.getElementById('xpText').innerText = `${userStats.xp}/${neededXP}`;
    }

    function addToNotebook(wordObj) {
      if (!wordObj) return;
      const exists = userStats.notebook.find(w => w.id === wordObj.id);
      if (!exists) {
        userStats.notebook.unshift(wordObj);
        updateNotebookBadge();
      }
    }

    function updateNotebookBadge() {
      const badge = document.getElementById('notebookBadge');
      if (userStats.notebook.length > 0) {
        badge.innerText = userStats.notebook.length;
        badge.classList.remove('hidden');
      } else {
        badge.classList.add('hidden');
      }
    }

      renderNotebookList();
      document.getElementById('notebookModal').classList.remove('hidden');
    }

      soundPop();
      document.getElementById('notebookModal').classList.add('hidden');

      container.innerHTML = '';

      if (userStats.notebook.length === 0) {

      });

    function removeFromNotebook(wordId) {
      userStats.notebook = userStats.notebook.filter(w => w.id !== wordId);
      soundPop();
      updateNotebookBadge();
      renderNotebookList();
    }

    function clearNotebook() {
      userStats.notebook = [];
      updateNotebookBadge();
      renderNotebookList();
      soundPop();
    }

    function triggerConfetti() {
      const canvas = document.getElementById('confettiCanvas');
      const ctx = canvas.getContext('2d');
      canvas.width = window.innerWidth;
      canvas.height = window.innerHeight;

      const particles = [];
      const colors = ['#f59e0b', '#10b981', '#06b6d4', '#ec4899', '#8b5cf6'];

      for (let i = 0; i < 60; i++) {
        particles.push({
          x: canvas.width / 2,
          y: canvas.height / 2,
          vx: (Math.random() - 0.5) * 12,
          vy: (Math.random() - 0.7) * 14,
          size: Math.random() * 6 + 3,
          color: colors[Math.floor(Math.random() * colors.length)],
          rotation: Math.random() * 360,
          rotationSpeed: (Math.random() - 0.5) * 8,
          gravity: 0.35,
          life: 1
        });
      }

      function animate() {
        ctx.clearRect(0, 0, canvas.width, canvas.height);
        let active = 0;
        particles.forEach(p => {
          if (p.life > 0) {
            active++;
            p.x += p.vx;
            p.y += p.vy;
            p.vy += p.gravity;
            p.rotation += p.rotationSpeed;
            p.life -= 0.02;

            ctx.save();
            ctx.translate(p.x, p.y);
            ctx.rotate((p.rotation * Math.PI) / 180);
            ctx.fillStyle = p.color;
            ctx.globalAlpha = Math.max(0, p.life);
            ctx.fillRect(-p.size / 2, -p.size / 2, p.size, p.size);
            ctx.restore();
          }
        });

        if (active > 0) requestAnimationFrame(animate);
        else ctx.clearRect(0, 0, canvas.width, canvas.height);
      }
      animate();
    }

    function shuffleArray(arr) {
      const copy = [...arr];
      for (let i = copy.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [copy[i], copy[j]] = [copy[j], copy[i]];
      }
      return copy;
    }

      const canvas = document.getElementById('confettiCanvas');
      if (canvas) {
        canvas.width = window.innerWidth;
        canvas.height = window.innerHeight;
      }
    });

      updateStatsUI();
